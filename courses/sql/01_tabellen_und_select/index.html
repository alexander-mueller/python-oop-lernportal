💡 1. Relationale Datenbanken & Tabellen

Daten werden in strukturierten Tabellen mit definierten Spaltentypen gespeichert:

CREATE TABLE kunden (
    id INTEGER PRIMARY KEY,
    name TEXT,
    guthaben REAL
);

INSERT INTO kunden (name, guthaben) VALUES ('Max Mustermann', 150.0);
SELECT * FROM kunden WHERE guthaben > 100;
